private enum class PState { LISTENING, NEW, DONE }

private data class PBook(
    val id: String,
    val title: String,
    val author: String,
    val genre: String,
    val state: PState,
    val percent: Float,
    /** The book's own total length — drives percent, remaining and finished. */
    val totalSeconds: Long,
    val chapter: String,
    val downloaded: Boolean,
    val source: String,
    /** «7 із 12» — a partial offline copy; null when the book is not partial. */
    val partial: String? = null
) {
    val entity: AudiobookEntity
        get() = AudiobookEntity(
            id = id,
            title = title,
            author = author,
            narrator = "Фікстура-читець",
            description = "Прототип-фікстура: $title.",
            coverDrawableRes = 0,
            coverImageUrl = null,
            genre = genre,
            sourceUrl = if (source == "Локальна") "" else "https://fixtures.4read.org/$id.html",
            isDownloaded = downloaded,
            totalDurationSeconds = totalSeconds,
            totalChapters = 12,
            rating = 4.5f
        )

    val percentLabel: String get() = "${(percent * 100).toInt()}%"

    /** Same arithmetic the production [LibraryBook] uses, same wording. */
    val remainingSeconds: Long get() = ((1f - percent) * totalSeconds).toLong()

    val remainingLabel: String get() = formatDuration(remainingSeconds)
    val totalLabel: String get() = formatDuration(totalSeconds)
}

/** «6 год 35 хв» / «48 хв» / «8 год» — the production remaining-time wording. */
private fun formatDuration(seconds: Long): String {
    if (seconds <= 0L) return "0 хв"
    val hours = seconds / 3_600L
    val minutes = (seconds % 3_600L) / 60L
    return when {
        hours > 0L && minutes > 0L -> "$hours год ${minutes.toString().padStart(2, '0')} хв"
        hours > 0L -> "$hours год"
        else -> "$minutes хв"
    }
}

/** Ukrainian plural for a book counter: 1 книга / 2 книги / 5 книг. */
private fun bookCount(n: Int): String = when {
    n % 10 == 1 && n % 100 != 11 -> "$n книга"
    n % 10 in 2..4 && n % 100 !in 12..14 -> "$n книги"
    else -> "$n книг"
}

/**
 * The offline marker the current screen already has and the redesign must not
 * lose: a cloud-with-check for a fully downloaded book, «7 із 12» while the
 * copy is still partial (#397).
 */
@Composable
private fun DownloadBadge(
    book: PBook,
    size: Dp = 16.dp,
    tint: Color = MaterialTheme.colorScheme.secondary
) {
    val partial = book.partial
    when {
        partial != null -> Text(
            text = partial,
            style = MaterialTheme.typography.labelSmall.copy(fontWeight = FontWeight.SemiBold),
            color = tint
        )
        book.downloaded -> Icon(
            imageVector = Icons.Default.CloudDone,
            contentDescription = "Завантажено",
            tint = tint,
            modifier = Modifier.size(size)
        )
    }
}

/** A cover with the listening progress drawn along its bottom edge. */
@Composable
private fun CoverWithProgress(book: PBook, width: Dp, height: Dp, radius: Dp) {
    Box(
        modifier = Modifier
            .size(width, height)
            .clip(RoundedCornerShape(radius))
    ) {
        BookCoverImage(
            book = book.entity,
            semantics = BookCoverSemantics.Decorative,
            modifier = Modifier.fillMaxSize()
        )
        if (book.state == PState.LISTENING) {
            LinearProgressIndicator(
                progress = { book.percent },
                modifier = Modifier
                    .align(Alignment.BottomCenter)
                    .fillMaxWidth()
                    .height(5.dp),
                color = MaterialTheme.colorScheme.primary,
                trackColor = Color.Black.copy(alpha = 0.35f)
            )
        }
    }
}

/** The prototype fixture as the production read model the card consumes. */
private fun PBook.toLibraryBook(): LibraryBook = LibraryBook(
    book = entity,
    progress = if (state == PState.NEW) {
        null
    } else {
        PlaybackProgressEntity(
            editionId = "edition-$id",

            currentChapterIndex = 4,

// ───────────────────────────────────────────────────────────────────────────
// VARIANT D — THE RECOMMENDATION: A as the base, B's «Нові» shelf, C's dense
// rows once a filter is on.
//
// One screen, one job order: resume (hero) → narrow down (one scrolling chip
// line) → the books, grouped by what they are to you. The four filter tabs are
// gone: «Закладки / Статистика / Люди» move to the ⋮ overflow, because they
// are not peers of «Книги» — that duplication was half the clutter.
//
// `filtered = true` renders the second half of the recommendation: with a
// status actively filtering, the screen drops the hero and the covers and
// becomes C's dense queue — several times more books per screen.
// ───────────────────────────────────────────────────────────────────────────

@Composable
private fun VariantDRecommended(books: List<PBook>, filtered: Boolean = false) {
    PhoneScreen {
        PrototypeTopBar(
            title = "Медіатека",
            subtitle = if (filtered) {
                "Фільтр: Слухаю · ${listening.size}"
            } else {
                "${bookCount(books.size)} · $libraryTotalLabel"
            }
        ) {
            SearchAction()
            IconButton(onClick = {}) {
                Icon(
                    imageVector = Icons.Default.MoreVert,
                    contentDescription = "Закладки, статистика, люди",
                    tint = MaterialTheme.colorScheme.onSurface
                )
            }
        }
        LazyColumn(
            modifier = Modifier.weight(1f),
            contentPadding = PaddingValues(bottom = AppDimens.SpaceSection)
        ) {
            if (!filtered) {
                item(key = "hero") { ContinueHeroCard(books.first { it.state == PState.LISTENING }) }
            }
            item(key = "chips") { StatusChipRow(books, selectedIndex = if (filtered) 1 else 0) }
            if (!filtered) {
                pSection("Слухаю", listening.drop(1), resume = true)
                item(key = "h-new") {
                    AppSectionHeader(title = "Нові", count = bookCount(fresh.size))
                }
                item(key = "shelf-new") {
                    LazyRow(
                        contentPadding = PaddingValues(horizontal = AppDimens.PageSides),
                        horizontalArrangement = Arrangement.spacedBy(AppDimens.SpaceMd)
                    ) {
                        items(fresh, key = { it.id }) { book -> ShelfTile(book, 116.dp) }
                    }
                }
                pSection("Завершені", finished, resume = false, done = true)
            } else {
                item(key = "h-dense") {
                    Row(
                        modifier = Modifier
                            .fillMaxWidth()
                            .padding(
                                start = AppDimens.PageSides,
                                end = AppDimens.PageSides,
                                top = AppDimens.SpaceLg,
                                bottom = AppDimens.SpaceSm
                            ),
                        verticalAlignment = Alignment.CenterVertically
                    ) {
                        Text(
                            text = "СЛУХАЮ · ${listening.size}",
                            style = MaterialTheme.typography.labelMedium.copy(
                                fontWeight = FontWeight.Bold,
                                letterSpacing = 1.sp
                            ),

/** One horizontally scrolling line of status chips — never wraps, never stacks. */
@Composable
private fun StatusChipRow(books: List<PBook>, selectedIndex: Int = 0) {
    val labels = listOf(
        "Усі ${books.size}",
        "Слухаю ${listening.size}",
        "Нові ${fresh.size}",
        "Завершені ${finished.size}",
        "Завантажені $downloadedCount"
    )
    LazyRow(
        modifier = Modifier.fillMaxWidth(),
        contentPadding = PaddingValues(horizontal = AppDimens.PageSides, vertical = AppDimens.SpaceSm),
        horizontalArrangement = Arrangement.spacedBy(AppDimens.SpaceSm)
    ) {
        items(labels.size) { index ->

        }
        item { Chip(label = "Фільтр", selected = false, trailingTune = true) }
    }
}

@Composable
private fun Chip(label: String, selected: Boolean, trailingTune: Boolean = false) {
    FilterChip(
        selected = selected,
        onClick = {},
        label = {
            Text(
                text = label,
                style = MaterialTheme.typography.labelMedium.copy(fontWeight = FontWeight.Bold)
            )
        },
        shape = RoundedCornerShape(AppDimens.RadiusPanel),
        leadingIcon = if (trailingTune) {
            {
                Icon(
                    imageVector = Icons.Default.Tune,
                    contentDescription = null,
                    modifier = Modifier.size(16.dp)
                )
            }
        } else null,
        colors = FilterChipDefaults.filterChipColors(
            selectedContainerColor = MaterialTheme.colorScheme.primary,
            selectedLabelColor = MaterialTheme.colorScheme.onPrimary,
            selectedLeadingIconColor = MaterialTheme.colorScheme.onPrimary,
            containerColor = MaterialTheme.colorScheme.surfaceContainerHigh,
            labelColor = MaterialTheme.colorScheme.onSurface
        ),
        border = FilterChipDefaults.filterChipBorder(
            enabled = true,
            selected = selected,
            borderColor = MaterialTheme.colorScheme.outlineVariant,
            selectedBorderColor = MaterialTheme.colorScheme.primary
        )
    )
}

private fun LazyListScope.pSection(
    title: String,
    books: List<PBook>,
    resume: Boolean,
    done: Boolean = false
) {
    item(key = "h-$title") {
        AppSectionHeader(title = title, count = bookCount(books.size))
    }
    items(books, key = { it.id }) { book ->
        PBookRow(book = book, resume = resume, done = done)
        HorizontalDivider(
            color = MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.4f),

@Composable
private fun PBookRow(book: PBook, resume: Boolean, done: Boolean) {
    BookRow(
        title = book.title,
        book = book.entity,
        author = book.author,
        stats = if (done) {
            "Прослухано повністю · ${book.totalLabel}"
        } else {
            "${book.chapter} · залишилось ${book.remainingLabel}"
        },
        progress = if (done) null else book.percent,
        badges = {
            Spacer(modifier = Modifier.width(AppDimens.SpaceSm))
            MetadataChip(source = book.source)
        },
        trailing = {
            Spacer(modifier = Modifier.width(AppDimens.SpaceMd))
            DownloadBadge(book)
            Spacer(modifier = Modifier.width(AppDimens.SpaceSm))
            when {
                resume -> FilledIconButton(onClick = {}, modifier = Modifier.size(44.dp)) {
                    Icon(
                        imageVector = Icons.Default.PlayArrow,
                        contentDescription = "Продовжити",
                        tint = MaterialTheme.colorScheme.onPrimary
                    )
                }
                done -> Icon(
                    imageVector = Icons.Default.CheckCircle,
                    contentDescription = "Завершено",
                    tint = MaterialTheme.colorScheme.secondary,
                    modifier = Modifier.size(24.dp)
                )
                else -> Icon(
                    imageVector = Icons.Default.PlayArrow,
                    contentDescription = "Слухати",
                    tint = MaterialTheme.colorScheme.primary,
                    modifier = Modifier.size(28.dp)
                )
            }
        },
        onClick = {}
    )
}

/** Queue row: rail, title, and a tabular counter on the right. */
@Composable
private fun QueueRow(book: PBook, done: Boolean = false) {
    val accent = if (book.state == PState.LISTENING) {
        MaterialTheme.colorScheme.primary
    } else {
        MaterialTheme.colorScheme.outlineVariant
    }
    Row(
        modifier = Modifier
            .fillMaxWidth()
            .height(IntrinsicSize.Min)
            .clickable {}
            .padding(horizontal = AppDimens.PageSides, vertical = AppDimens.SpaceSm),
        verticalAlignment = Alignment.CenterVertically
    ) {
        // The timeline rail: one hairline per row plus a fill dot on the
        // in-progress book. Stacked rows read as one unbroken thread.
        Box(
            modifier = Modifier
                .width(24.dp)
                .fillMaxHeight(),
            contentAlignment = Alignment.Center
        ) {
            Box(
                modifier = Modifier
                    .width(2.dp)
                    .fillMaxHeight()
                    .background(MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.5f))
            )
            Box(
                modifier = Modifier
                    .size(if (book.state == PState.LISTENING) 12.dp else 8.dp)
                    .clip(CircleShape)
                    .background(accent)
            )
        }
        Spacer(modifier = Modifier.width(AppDimens.SpaceSm))
        Column(modifier = Modifier.weight(1f)) {
            Text(
                text = book.title,
                style = MaterialTheme.typography.titleSmall.copy(fontWeight = FontWeight.Bold),
                color = if (done) {
                    MaterialTheme.colorScheme.onSurfaceVariant
                } else {
                    MaterialTheme.colorScheme.onSurface
                },
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
            Text(
                text = if (done) "${book.author} · прослухано" else "${book.author} · ${book.chapter}",
                style = MaterialTheme.typography.labelSmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant,
                maxLines = 1,
                overflow = TextOverflow.Ellipsis
            )
        }
        Spacer(modifier = Modifier.width(AppDimens.SpaceMd))
        Column(horizontalAlignment = Alignment.End) {
            Text(
                text = if (done) book.totalLabel else book.remainingLabel,
                style = MaterialTheme.typography.labelLarge.copy(fontWeight = FontWeight.Bold),
                color = if (book.state == PState.LISTENING) {
                    MaterialTheme.colorScheme.primary
                } else {
                    MaterialTheme.colorScheme.onSurfaceVariant
                }
            )
            Text(
                text = when {
                    done -> "готово"
                    book.state == PState.NEW -> "новий"
                    else -> book.percentLabel
                },
                style = MaterialTheme.typography.labelSmall,
                color = MaterialTheme.colorScheme.onSurfaceVariant
            )
        }
        Spacer(modifier = Modifier.width(AppDimens.SpaceSm))
        DownloadBadge(book, size = 15.dp)
        Spacer(modifier = Modifier.width(AppDimens.SpaceSm))
        Icon(
            imageVector = if (done) Icons.Default.CheckCircle else Icons.Default.PlayArrow,
            contentDescription = if (done) "Завершено" else "Слухати",
            tint = if (done) MaterialTheme.colorScheme.secondary else MaterialTheme.colorScheme.primary,
            modifier = Modifier.size(22.dp)
